Full setup guide 
// In Claude Code. No path, no config.
/plugin marketplace add ronak-create/FableCut
/plugin install fablecut@fablecut

Download the bundle and open it. Claude Desktop installs the server for you, no path needed.

Download fablecut.mcpb 
codex mcp add fablecut -- \
  npx -y fablecut mcp
gemini mcp add -s user fablecut \
  npx -y fablecut mcp
// ~/.cursor/mcp.json
{
  "mcpServers": {
    "fablecut": {
      "command": "npx",
      "args": ["-y", "fablecut", "mcp"]
    }
  }
}
// opencode.json
{
  "mcp": {
    "fablecut": {
      "type": "local",
      "command": ["npx", "-y", "fablecut", "mcp"],
      "enabled": true
    }
  }
}
amp mcp add fablecut -- \
  npx -y fablecut mcp
